# tools/dist_units_ausnahmen_win64.txt
#
# Units, die im win64-Auslieferbaum (tools/build_dist.sh win64) FEHLEN DUERFEN,
# je mit Grund. Format:
#
#   <pfad/zur/quelle.lyx>  # <Grund, mit Issue-Nummer>
#
# Alles, was hier NICHT steht und trotzdem kein .lyu hat, macht den Lauf rot.
# Der Eintrag entschuldigt die Luecke nicht, er macht sie nur benennbar: jede
# Zeile hier gehoert in UNITS-FEHLER.txt des Baums und damit in die Hand des
# Paketbauers.
#
# Stand 2026-11-10, lyxc 1.2.34B: 16 von 500 Units, drei Ursachen.

# ── #2459: geschachtelter Feldzugriff auf einen importierten Typ ─────────────
# `r.min.x` meldet unter win64 "error: #2134: Feld nicht aufloesbar (Typ der
# Basis unbekannt)", fuer linux uebersetzt dieselbe Quelle. Kein Pfad- oder
# Aufloesungsproblem der .lyu — es ist eine unvollstaendige Aufzaehlung in der
# Typbestimmung.
data/binary.lyx         # #2459 win64-Feldzugriff (Zeile 38, dtype)
data/csv.lyx            # #2459 win64-Feldzugriff (Zeile 115, field)
data/json.lyx           # #2459 win64-Feldzugriff (Zeile 107, field)
data/ops.lyx            # #2459 win64-Feldzugriff (Zeile 43, field)
data/print.lyx          # #2459 win64-Feldzugriff (Zeile 81, field)
std/circle.lyx          # #2459 win64-Feldzugriff (Zeile 367, r.min.x)
std/net/socket.lyx      # #2459 win64-Feldzugriff (Zeile 733, a)

# ── #2460: mulhi/umulhi fehlen in der ir_lower-Aufzaehlung ───────────────────
# Der direkte x86-Codegen kennt beide (codegen_x86.lyx:21119/21130), ir_lower
# nicht — und win64 geht ueber ir_lower. Folge: keine Krypto auf win64.
std/crypto/bignum.lyx   # #2460 umulhi auf win64 unbekannt
std/crypto/ed25519.lyx  # #2460 umulhi auf win64 unbekannt
std/crypto/rsa_native.lyx # #2460 umulhi auf win64 unbekannt
std/crypto/x509.lyx     # #2460 umulhi auf win64 unbekannt
std/net/tls13.lyx       # #2460 umulhi auf win64 unbekannt

# ── #1678: Linux-Syscalls ohne Windows-Ersatz ───────────────────────────────
# Diese drei lehnt der Compiler LAUT ab, und das ist richtig: sys_select und
# sys_arch_prctl haben auf win64 keine Entsprechung, und eine erfundene waere
# der stille Default. Sie verschwinden erst, wenn #1678 den Syscall-Ersatz
# bringt — dieselbe Luecke, die auch die Warnung beim Compilerbau erzeugt.
std/db/postgres.lyx     # #2519 win64 uebersetzt Linux-Syscalls STILL (syscall-Befehl im PE); die Begruendung #1678 trifft nicht mehr zu
std/net/ssh.lyx         # dito -- #2519
std/net/syscalls.lyx    # dito -- #2519
std/security_ext_x86.lyx # arch_prctl ist x86-eigen; win64 weist es benannt ab -- #2512
